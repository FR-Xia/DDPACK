cmake_minimum_required(VERSION 3.20)
project(DDPACK VERSION 1.1.0 LANGUAGES CXX)
include(CTest)
include(GNUInstallDirs)
option(DDPACK_BUILD_EXAMPLES "Build the native example runner" ON)
# Use a packaged Eigen or an explicitly supplied source tree; no hidden downloads.
set(DDPACK_EIGEN_DIR "" CACHE PATH "Directory containing Eigen/Core")
if(DDPACK_EIGEN_DIR)
  set(DDPACK_EIGEN_INCLUDE "${DDPACK_EIGEN_DIR}")
  if(NOT EXISTS "${DDPACK_EIGEN_INCLUDE}/Eigen/Core")
    message(FATAL_ERROR "DDPACK_EIGEN_DIR must contain Eigen/Core")
  endif()
else()
  find_package(Eigen3 3.4 QUIET NO_MODULE)
endif()
if(NOT TARGET Eigen3::Eigen)
  find_path(DDPACK_EIGEN_INCLUDE Eigen/Core HINTS "${DDPACK_EIGEN_DIR}" PATH_SUFFIXES eigen3)
  if(NOT DDPACK_EIGEN_INCLUDE)
    message(FATAL_ERROR "Install Eigen >= 3.4 or set DDPACK_EIGEN_DIR to its source directory")
  endif()
  add_library(Eigen3::Eigen INTERFACE IMPORTED)
  set_target_properties(Eigen3::Eigen PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${DDPACK_EIGEN_INCLUDE}")
endif()
add_library(ddpack cpp/src/models.cpp cpp/src/solver.cpp cpp/src/diagnostics.cpp cpp/src/refinement.cpp cpp/src/line_search.cpp cpp/src/numerics.cpp cpp/src/derivatives.cpp)
add_library(DDPACK::ddpack ALIAS ddpack)
target_compile_features(ddpack PUBLIC cxx_std_17)
target_include_directories(ddpack PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/cpp/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(ddpack PUBLIC Eigen3::Eigen)
if(MSVC)
  target_compile_options(ddpack PRIVATE /W4 /permissive- /utf-8 /bigobj)
else()
  target_compile_options(ddpack PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(DDPACK_BUILD_EXAMPLES OR BUILD_TESTING)
  add_library(ddpack_cases STATIC cpp/examples/problems.cpp)
  target_link_libraries(ddpack_cases PUBLIC DDPACK::ddpack)
  target_include_directories(ddpack_cases PUBLIC cpp/examples)
endif()
if(DDPACK_BUILD_EXAMPLES)
  add_executable(ddpack_examples cpp/examples/main.cpp)
  target_link_libraries(ddpack_examples PRIVATE ddpack_cases)
  add_executable(ddpack_benchmark cpp/benchmarks/benchmark.cpp)
  target_link_libraries(ddpack_benchmark PRIVATE ddpack_cases)
endif()
if(BUILD_TESTING)
  add_executable(ddpack_tests cpp/tests/test_ddpack.cpp)
  target_link_libraries(ddpack_tests PRIVATE ddpack_cases)
  add_test(NAME native_contracts COMMAND ddpack_tests)
  add_executable(ddpack_parity cpp/tests/test_parity.cpp)
  target_link_libraries(ddpack_parity PRIVATE ddpack_cases)
  add_test(NAME matlab_parity COMMAND ddpack_parity cpp/tests/reference)
  set_tests_properties(matlab_parity PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME matlab_source_integrity COMMAND "${CMAKE_COMMAND}" -DROOT=${CMAKE_CURRENT_SOURCE_DIR} -P "${CMAKE_CURRENT_SOURCE_DIR}/cpp/cmake/check_reference_sources.cmake")
  add_executable(ddpack_cli_tests cpp/tests/test_cli_arguments.cpp)
  target_compile_features(ddpack_cli_tests PRIVATE cxx_std_17)
  target_include_directories(ddpack_cli_tests PRIVATE cpp/examples)
  add_test(NAME cli_argument_encoding COMMAND ddpack_cli_tests)
  add_executable(ddpack_line_search_tests cpp/tests/test_line_search.cpp)
  target_link_libraries(ddpack_line_search_tests PRIVATE ddpack_cases)
  target_include_directories(ddpack_line_search_tests PRIVATE cpp/src)
  add_test(NAME actual_merit_line_search COMMAND ddpack_line_search_tests)
  add_executable(ddpack_numerics_tests cpp/tests/test_numerics.cpp)
  target_link_libraries(ddpack_numerics_tests PRIVATE ddpack_cases)
  target_include_directories(ddpack_numerics_tests PRIVATE cpp/src)
  add_test(NAME numerical_robustness COMMAND ddpack_numerics_tests)
  add_executable(ddpack_derivative_tests cpp/tests/test_derivatives.cpp)
  target_link_libraries(ddpack_derivative_tests PRIVATE ddpack_cases)
  add_test(NAME derivative_contracts COMMAND ddpack_derivative_tests)
  if(TARGET ddpack_examples AND TARGET ddpack_benchmark)
    add_test(NAME unicode_cli COMMAND "${CMAKE_COMMAND}"
      "-DSOURCE_ROOT=${CMAKE_CURRENT_SOURCE_DIR}" "-DOUTPUT_ROOT=${CMAKE_BINARY_DIR}/Testing"
      "-DEXAMPLES=$<TARGET_FILE:ddpack_examples>" "-DPARITY=$<TARGET_FILE:ddpack_parity>"
      "-DBENCHMARK=$<TARGET_FILE:ddpack_benchmark>"
      -P "${CMAKE_CURRENT_SOURCE_DIR}/cpp/cmake/check_unicode_cli.cmake")
    set_tests_properties(unicode_cli PROPERTIES TIMEOUT 300)
  endif()
endif()
# MinGW needs the Unicode CRT startup when a program defines wmain.
if(WIN32 AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  foreach(cli_target ddpack_examples ddpack_benchmark ddpack_parity ddpack_cli_tests ddpack_derivative_tests)
    if(TARGET ${cli_target})
      target_link_options(${cli_target} PRIVATE -municode)
    endif()
  endforeach()
endif()
install(TARGETS ddpack EXPORT DDPACKTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY cpp/include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT DDPACKTargets NAMESPACE DDPACK:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/DDPACK)
include(CMakePackageConfigHelpers)
configure_package_config_file(cpp/cmake/DDPACKConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/DDPACKConfig.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/DDPACK)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/DDPACKConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/DDPACKConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/DDPACKConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/DDPACK)
